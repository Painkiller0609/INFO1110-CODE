def match_clubs(student,clubs):
    Score = 0
    matches = []
    for club in clubs:
      
      if student["main_interest"] == club["category"]: 
        Score = Score + 3

      for keywords in club["keywords"]:
            if keywords in  student["hobbies"]:
              Score = Score + 1

      if student["social_score"] >= 60 and "socail" in club["keywords"]:
                Score = Score + 1
      if student["skill_score"] >= 60 and "skills" in club["keywords"] or "career" in club["keywords"] or "study" in club["keywords"]:
                Score = Score + 1
      
        match = {
          "name": club["name"],
          "score": Score
        }     
        matches.append(match)
    score_list = []
    for match2 in matches:
      score_list.append([
        -match2["score"],
        match2["name"],
        match2["score"]
      ])
    score_list.sort()
    first_matches = score_list[:2]

    filename = student["name"] + ".txt"
    file = open(filename,"w")

    file.write("First Club Recommendation\n")
    file.write('Name:"'+first_matches[1][1]+'"\n')
    file.write("Score:"+ str(first_matches[1][2]+"\n"))
    file.close()

    return [first_matches[0][1],first_matches[1][1]]
